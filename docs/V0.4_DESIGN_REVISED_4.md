interface VirtualBuyPlan {

## 5. 다음날 보유 종목 AI 의견 (사용자 명세 §5·§6)

### 5-1. 보유 의견 인터페이스

```ts
type HoldingVerdict = 'HOLD' | 'PARTIAL_SELL' | 'SELL' | 'WATCH_MORE' | 'WARN_SELL';

interface HoldingOpinion {
  evaluated_at: string;
  current_price: number;
  pnl_pct: number;
  verdict: HoldingVerdict;
  verdict_label: string;       // 사용자 노출 ("보유" / "일부매도" / "매도" / "추가 관찰" / "매도 주의")
  reason: string;              // 자연어 1~2 문장
  today_actions: string[];     // 오늘 행동 가이드 체크리스트 3개 내외
}
```

### 5-2. 판정 로직 (사용자 명세 §6 단순 매핑)

```ts
function buildHoldingOpinion(
  record: VirtualBuyRecord,
  currentPrice: number,
): HoldingOpinion {
  const plan = record.scoring_snapshot.buy_plan;
  const pnl_pct = (currentPrice - plan.first_buy_price) / plan.first_buy_price * 100;

  // 사용자 명세 6 단계 매핑
  if (currentPrice >= plan.target_sell_price) {
    return {
      verdict: 'PARTIAL_SELL',
      verdict_label: '일부매도',
      reason: `목표 매도가(${plan.target_sell_price.toLocaleString()}원)에 도달했습니다 — 일부 차익 실현 검토.`,
      today_actions: ['일부 차익실현 검토', '남은 수량 추세 확인', '추가 진입 금지'],
      pnl_pct, current_price: currentPrice, evaluated_at: nowISO(),
    };
  }

  if (currentPrice <= plan.stop_loss_price) {
    return {
      verdict: 'SELL',
      verdict_label: '매도',
      reason: `손절 기준가(${plan.stop_loss_price.toLocaleString()}원) 이탈 — 매도 검토 우선. 손실 확대 방지.`,
      today_actions: ['보유 이유 재점검', '손실 확대 방지', '추가매수 절대 금지'],
      pnl_pct, current_price: currentPrice, evaluated_at: nowISO(),
    };
  }

  if (pnl_pct >= 5) {
    return {
      verdict: 'HOLD',                       // 또는 PARTIAL_SELL — 사용자 명세 "보유 또는 일부매도"
      verdict_label: '보유 (일부매도 검토 가능)',
      reason: `+${pnl_pct.toFixed(1)}% 상승 중이지만 목표가에는 미도달. 보유 또는 일부매도 모두 가능합니다.`,
      today_actions: ['목표 매도가 유지', '추세 유지 확인', '거래량 감소 여부 확인'],
      pnl_pct, current_price: currentPrice, evaluated_at: nowISO(),
    };
  }

  if (pnl_pct <= -5) {
    return {
      verdict: 'WARN_SELL',
      verdict_label: '매도 주의',
      reason: `${pnl_pct.toFixed(1)}% 하락. 손절 기준 근접 — 매도 주의.`,
      today_actions: ['손절 기준 재점검', '추가매수 금지', '하락 가속 여부 확인'],
      pnl_pct, current_price: currentPrice, evaluated_at: nowISO(),
    };
  }

  // 그 외 — 보유 (사용자 명세 "그 외: 보유")
  return {
    verdict: 'HOLD',
    verdict_label: '보유',
    reason: `매수가 대비 ${pnl_pct >= 0 ? '+' : ''}${pnl_pct.toFixed(1)}% — 목표가/손절가 모두 미도달. 하루 더 보유 관찰 가능.`,
    today_actions: ['추가매수하지 않음', '목표 매도가 유지', '거래량 감소 여부 확인'],
    pnl_pct, current_price: currentPrice, evaluated_at: nowISO(),
  };
}
```

## 12. 주의사항

### 12-1. 자동매매 절대 금지 (사용자 명세 §9 재확인)
- 키움 API 호출 0건
- 모든 가상 기능 = localStorage + read-only fetch
- "매수" / "매도" 라벨은 **개인 기록용 AI 판단** — 항상 안내 동반

### 12-2. 기본값은 "제안" 일 뿐 (사용자 명세)
- 폼의 모든 필드는 사용자가 직접 수정 가능
- 자동 제안값은 회색으로 표시, 사용자 입력 시 강조 변경
- 비워두면 폼 차단 (수치 필수 4종: 1차 매수가, 목표 매도가, 손절 기준가)

### 12-3. ②③ 카테고리 매수 시 (v0.4 한계)
- scan_dump.candidates_bottom 미수록 → 자동 제안 불가
- 폼에 안내: "②③ 카테고리는 사용자가 직접 1차 매수가 / 목표 매도가 / 손절 기준가 를 입력해 주세요. v0.4.1 에서 자동 제안 예정."
- 사용자가 직접 입력하면 기록 가능

### 12-4. 매수가 매도가 손절가 음수/0 입력 차단
- `first_buy_price > 0` 필수
- `target_sell_price > first_buy_price` 권장 (예외 허용 — 사용자 의사 존중)
- `stop_loss_price < first_buy_price` 권장
- 위 권장 사항 위반 시 폼에 경고 띄우되 저장은 허용

### 12-5. 가상 보유 의견 자동 업데이트
- 카드 진입 시점에 client side fetch (`useEffect` 안에서 scan_dump 의 close 매칭)
- 매시간 자동 새로고침 안 함 (사용자 명시적 새로고침 또는 페이지 진입 시만)

### 12-6. localStorage 불변성
- `scoring_snapshot.buy_plan` 은 저장 후 **불변**
- `holding_opinion` 은 매번 새로 계산 (저장하지 않음 — 현재가 변동)
- `evaluations.current` 는 카드 진입 시점에 매번 계산

### 12-7. enum 키 변경 영향
- `BUY_SPLIT` → `BUY` 변경은 v0.4 첫 출시이므로 기존 사용자 데이터 0건
- v0.4.1+ 부터는 enum 키 안정화
